# Task API

The Task API provides Swift-friendly access to Mach tasks, which represent execution contexts and address spaces.

## Core Types


## Usage Example

```swift
import Mach

// Get the current task
let task = Mach.Task.current

// Get basic info
let basicInfo = try task.info.basicInfo
print("Virtual size: \(basicInfo.virtual_size)")
print("Resident size: \(basicInfo.resident_size)")

// Get the task's PID
let pid = try task.pid
print("Task PID: \(pid)")

// Get all threads
let threads = try task.threads
print("Thread count: \(threads.count)")

// Suspend and resume
try task.suspend()
defer { try? task.resume() }

// Work with task role
let role = try task.role
print("Current role: \(role)")
try task.setRole(to: .background)

// Get special ports
let bootstrap = try task.getSpecialPort(.bootstrap, as: Mach.Port.self)
let hostPort = try task.hostPort

// Memory info
let vmInfo = try task.info.vmInfo
print("Page size: \(vmInfo.page_size)")
```
