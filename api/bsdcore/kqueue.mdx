## Overview

The BSDCore KQueue API provides access to the kernel event notification mechanism (kqueue). KQueue is a scalable event notification interface for monitoring file descriptors, processes, signals, timers, and other system events.

## Example Usage

<CodeGroup>

```swift Basic KQueue
import BSDCore
import Darwin

// Create a kqueue
let kq = try BSD.KQueue()

// Monitor a file descriptor for read events
let fd: Int32 = // ... your file descriptor
let event = kevent(
    identifier: UInt(fd),
    filter: .read,
    flags: [.add, .enable],
    filterFlags: 0,
    filterData: 0,
    userData: nil
)

// Register the event
try kq.event(registeringEvents: [event])

// Wait for events (with timeout)
var timeout = timespec(tv_sec: 5, tv_nsec: 0)
if let events = try kq.event(
    retrievingEventsOfCount: 10,
    timeout: &timeout
) {
    for event in events {
        print("Event triggered on fd \(event.ident)")
    }
}
```

```swift Monitor File Changes
import BSDCore
import Darwin

// Open a file
let path = "/path/to/file.txt"
let fd = open(path, O_RDONLY)
defer { close(fd) }

// Create kqueue
let kq = try BSD.KQueue()

// Monitor for file modifications
let event = kevent(
    identifier: UInt(fd),
    filter: .vnode,
    flags: [.add, .enable, .clear],
    filterFlags: UInt32(NOTE_WRITE | NOTE_DELETE | NOTE_RENAME),
    filterData: 0,
    userData: nil
)

try kq.event(registeringEvents: [event])

print("Monitoring \(path) for changes...")

while true {
    if let events = try kq.event(retrievingEventsOfCount: 1) {
        for event in events {
            if event.fflags & UInt32(NOTE_WRITE) != 0 {
                print("File was modified")
            }
            if event.fflags & UInt32(NOTE_DELETE) != 0 {
                print("File was deleted")
                break
            }
            if event.fflags & UInt32(NOTE_RENAME) != 0 {
                print("File was renamed")
            }
        }
    }
}
```

```swift Timer Events
import BSDCore
import Darwin

let kq = try BSD.KQueue()

// Create a timer that fires every 2 seconds
let timerID: UInt = 1
let event = kevent(
    identifier: timerID,
    filter: .timer,
    flags: [.add, .enable],
    filterFlags: 0,
    filterData: 2000,  // milliseconds
    userData: nil
)

try kq.event(registeringEvents: [event])

print("Timer started...")

for _ in 0..<5 {
    if let events = try kq.event(retrievingEventsOfCount: 1) {
        for event in events {
            print("Timer fired! Count: \(event.data)")
        }
    }
}
```

```swift Multiple Event Types
import BSDCore
import Darwin

let kq = try BSD.KQueue()

// Monitor multiple types of events
var events = [kevent]()

// Timer event
events.append(kevent(
    identifier: 1,
    filter: .timer,
    flags: [.add, .enable],
    filterFlags: 0,
    filterData: 1000,  // 1 second
    userData: nil
))

// Process event
events.append(kevent(
    identifier: UInt(getpid()),
    filter: .proc,
    flags: [.add, .enable],
    filterFlags: UInt32(NOTE_SIGNAL),
    filterData: 0,
    userData: nil
))

// Register all events
try kq.event(registeringEvents: events)

// Wait for any event
while true {
    if let triggeredEvents = try kq.event(retrievingEventsOfCount: 10) {
        for event in triggeredEvents {
            switch BSD.KEventFilterType(rawValue: event.filter).name {
            case "timer":
                print("Timer event")
            case "proc":
                print("Process event")
            default:
                print("Other event")
            }
        }
    }
}
```

## Notes

- KQueues are automatically closed when the object is deallocated
- Events can be added, modified, or deleted in a single `event()` call
- Timeouts are optional; omitting them causes the call to block indefinitely
- The `oneshot` flag automatically removes an event after it triggers
- The `clear` flag resets edge-triggered events after retrieval
- Each filter type has specific filter flags (e.g., `NOTE_WRITE` for vnode events)
- User data pointers allow associating context with events
