## Overview

The Linking module provides a clean Swift API for dynamic library loading using `dlopen` and `dlsym`. It simplifies accessing frameworks, loading symbols, and working with Objective-C classes at runtime.

<Note>
**Source Location**: `Sources/Linking/`

**Dependencies**: None (standalone module)

**Future**: May be integrated into BSDCore in future versions
</Note>

## Core Components

### Handle Types

The module provides three main handle types:

```swift linking.swift:3-26
// Base handle class
public class Handle: @unchecked Sendable {
    public enum HandleType {
        case library
        case symbol
    }
    
    public let name: String
    public let rawHandle: UnsafeMutableRawPointer
    public let type: HandleType
}

// Library handle (extends Handle)
public class Library: Handle

// Symbol handle (extends Handle)
public class Symbol: Handle

// Framework handle (extends Library)
public class Framework: Library
```

func loadLibrary(named name: String) -> Library? {
    // Try common library paths
    let paths = [
        "/usr/lib/lib\(name).dylib",
        "/usr/local/lib/lib\(name).dylib",
        "/opt/homebrew/lib/lib\(name).dylib"
    ]
    
    for path in paths {
        if let lib = Library(path: path) {
            return lib
        }
    }
    
    return nil
}

}

### Lazy Loading Pattern

```swift
import Linking

class LazyLibrary {
    private var _framework: Framework?
    
    var framework: Framework? {
        if _framework == nil {
            _framework = Framework("SomeFramework")
        }
        return _framework
    }
    
    lazy var someFunction: (@convention(c) (Int) -> Int)? = {
        framework?.get(symbol: "some_function")?.cast()
    }()
    
    func callFunction(_ value: Int) -> Int? {
        return someFunction?(value)
    }
}

let lib = LazyLibrary()
if let result = lib.callFunction(42) {
    print("Result: \(result)")
}
```

### Global Symbol Cache

```swift
import Linking

class SymbolCache {
    static let shared = SymbolCache()
    
    private var libraries: [String: Library] = [:]
    private var symbols: [String: Symbol] = [:]
    
    func library(named name: String, isPrivate: Bool = false) -> Library? {
        let key = "\(name)_\(isPrivate)"
        if let cached = libraries[key] {
            return cached
        }
        
        guard let lib = Framework(name, isPrivate: isPrivate) else {
            return nil
        }
        
        libraries[key] = lib
        return lib
    }
    
    func symbol(named name: String, in libraryName: String) -> Symbol? {
        let key = "\(libraryName):\(name)"
        if let cached = symbols[key] {
            return cached
        }
        
        guard let lib = library(named: libraryName),
              let symbol = lib.get(symbol: name) else {
            return nil
        }
        
        symbols[key] = symbol
        return symbol
    }
}

// Usage
if let symbol = SymbolCache.shared.symbol(named: "malloc", in: "System") {
    let malloc: @convention(c) (Int) -> UnsafeMutableRawPointer? = symbol.cast()
    let ptr = malloc(1024)
    // Use pointer...
}
```

import Linking
import Foundation

protocol Plugin {
    var name: String { get }
    func execute()
}

class PluginLoader {
    func loadPlugin(at path: String) -> Plugin? {
        guard let lib = Library(path: path) else {
            return nil
        }
        
        // Get plugin initialization function
        guard let initSymbol = lib.get(symbol: "plugin_init") else {
            return nil
        }
        
        typealias InitFunc = @convention(c) () -> UnsafeMutableRawPointer?
        let initFunc: InitFunc = initSymbol.cast()
        
        guard let pluginPtr = initFunc() else {
            return nil
        }
        
        // Convert to Swift plugin object
        // Note: Requires proper plugin ABI design
        return Unmanaged<AnyObject>.fromOpaque(pluginPtr)
            .takeRetainedValue() as? Plugin
    }
    
    func loadPlugins(in directory: String) -> [Plugin] {
        let fm = FileManager.default
        guard let files = try? fm.contentsOfDirectory(atPath: directory) else {
            return []
        }
        
        return files
            .filter { $0.hasSuffix(".dylib") }
            .compactMap { loadPlugin(at: "\(directory)/\($0)") }
    }
}

let loader = PluginLoader()
let plugins = loader.loadPlugins(in: "/path/to/plugins")
plugins.forEach { $0.execute() }

## Thread Safety

<Warning>
- Library loading is thread-safe (handled by dyld)
- Symbol lookups are thread-safe
- Handle objects are marked `@unchecked Sendable`
- Concurrent access to the same library is safe
- Function pointers obtained from symbols should be used carefully across threads
</Warning>

## Best Practices

1. **Cache library handles**: Loading libraries is expensive
2. **Check for nil**: Symbol lookups can fail
3. **Use proper function types**: Incorrect casts lead to crashes
4. **Release handles**: Libraries are automatically closed on dealloc
5. **Validate symbol addresses**: When using address-based loading
6. **Handle versioning**: Framework APIs can change between OS versions

## Performance Considerations

```swift
// Good: Load once, use many times
let lib = Framework("Foundation")!
let symbol1 = lib.get(symbol: "func1")
let symbol2 = lib.get(symbol: "func2")

// Bad: Repeated loading
for _ in 0..<1000 {
    let lib = Framework("Foundation")  // Expensive!
    let symbol = lib?.get(symbol: "func1")
}

// Good: Cache symbols
class SymbolCache {
    let malloc: (@convention(c) (Int) -> UnsafeMutableRawPointer?)?
    let free: (@convention(c) (UnsafeMutableRawPointer) -> Void)?
    
    init() {
        let lib = Library(path: "/usr/lib/libSystem.B.dylib")
        malloc = lib?.get(symbol: "malloc")?.cast()
        free = lib?.get(symbol: "free")?.cast()
    }
}
```

## Limitations

- No automatic dependency resolution
- Limited support for C++ symbols (name mangling)
- Objective-C runtime interaction requires additional work
- No versioning support
- No unload functionality (relies on automatic dealloc)
